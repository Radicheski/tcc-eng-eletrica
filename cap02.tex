\chapter{OBJETIVOS} \label{cha:objetivos}

\begin{itemize}
    \item Projetar e construir um testador de bateria capaz de controlar ciclos de carga e descarga.
    \item Implementar um sistema de medição para estimação da resistência interna da bateria em tempo real.
    \item Criar um software de aquisição dos dados em um computador.
    \item Garantir que o sistema respeite os limites de segurança da bateria durante os testes.
\end{itemize}
